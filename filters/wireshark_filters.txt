# ================================
# Wireshark Filters - SOC Analysis
# ================================

# HTTP Traffic
http

# DNS Traffic
dns

# Filter Traffic by Specific IP
ip.addr == 192.168.1.1

# Unusual Ports (Not HTTP/HTTPS)
tcp.port != 80 && tcp.port != 443

# Suspicious High Ports (Common in attacks)
tcp.port == 4444 || tcp.port == 8080

# DNS Query Only
dns.flags.response == 0

# DNS Response Only
dns.flags.response == 1

# TCP SYN Packets (Possible Scan)
tcp.flags.syn == 1 && tcp.flags.ack == 0

# ICMP Traffic (Ping)
icmp

# Failed Connections / Reset Packets
tcp.flags.reset == 1

# FTP Traffic (Unsecure)
ftp

# Telnet Traffic (Unsecure)
telnet
